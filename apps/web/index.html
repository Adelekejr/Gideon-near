    03 / ONE CALL

    Every check leaves a receipt.

    Each result carries a sha256 receipt over exactly what was checked. With a signing key it’s also ed25519-signed, so a wallet or agent can prove it checked before it sent.

    // before the send
import { check } from "@gideon/core"
import { liveSources } from "@gideon/sources"

const r = await check({
  to: "89a46298…f4126944", network: "mainnet",
  destination: "exchange", exchange: "binance"
}, liveSources())
// → verdict "BLOCKED", headline.code "TX_HASH_AS_ACCOUNT"
//   "This is a Tron transaction hash, not an account"
if (r.verdict !== "SAFE") return r.headline
